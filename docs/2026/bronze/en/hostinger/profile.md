### 5.3 Quickstart Code Example

```python
"""
Junction 2026: Manage cloud infrastructure and configure DNS in 3 minutes via the Hostinger API
Specification: OpenAPI / Bearer Token Authentication
"""
import os
import requests

HOSTINGER_API_TOKEN = os.environ.get("HOSTINGER_API_TOKEN", "your_hostinger_api_token")
BASE_URL = "https://api.hostinger.com/v1"
HEADERS = {
    "Authorization": f"Bearer {HOSTINGER_API_TOKEN}",
    "Content-Type": "application/json"
}

def list_virtual_servers():
    """Retrieve operational status for all VPS instances under the account."""
    endpoint = f"{BASE_URL}/vps"
    print("[*] Querying Hostinger Cloud for active VPS instances...")
    
    try:
        response = requests.get(endpoint, headers=HEADERS, timeout=10)
        if response.status_code == 200:
            servers = response.json().get("data", [])
            print(f"[+] Successfully retrieved {len(servers)} VPS instance(s):")
            for server in servers:
                print(f" - ID: {server.get('id')}, Name: {server.get('name')}, IP: {server.get('ip')}, Status: {server.get('status')}")
            return servers
        else:
            print(f"[!] API returned status code: {response.status_code} (Check HOSTINGER_API_TOKEN permissions)")
            return []
    except Exception as e:
        print(f"[-] Connection error: {e}")
        return []

def configure_dns_record(domain: str, record_type: str, name: str, content: str):
    """Add a public DNS subdomain record for a hackathon project."""
    endpoint = f"{BASE_URL}/dns/{domain}/records"
    payload = {
        "type": record_type,
        "name": name,
        "content": content,
        "ttl": 300
    }
    print(f"[*] Creating DNS record for {domain}: {name}.{domain} -> {content}...")
    try:
        response = requests.post(endpoint, headers=HEADERS, json=payload, timeout=10)
        if response.status_code in [200, 201]:
            print(f"[✓] DNS record configured successfully! Active globally.")
        else:
            print(f"[!] Configuration response: {response.text}")
    except Exception as e:
        print(f"[-] Request failed: {e}")

if __name__ == "__main__":
    print("=== Hostinger Cloud Infrastructure Quickstart ===")
    list_virtual_servers()
    # Example: configure_dns_record("my-hackathon-project.com", "A", "api", "192.0.2.1")
```
